# From-Expression Syntax Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement `/expression` syntax for iterate-and-assign, replacing the old DRL `from` keyword with OOPath-style from-expressions.

**Architecture:** New grammar rule `boundFromExpression` alongside existing `boundOopath`. New `FromExpressionIR` record in the IR. New `FromExpressionDataProvider` class implementing `DataProvider` to evaluate a compiled MVEL3 lambda and iterate results. Lambda compilation reuses the existing `DrlxLambdaCompiler` batch-compile infrastructure.

**Tech Stack:** ANTLR4, MVEL3, protobuf, drools-base (`From`, `DataProvider`, `Pattern`, `Declaration`)

**Spec:** `specs/2026-08-17-95-from-expression-design.md`

```java
package org.drools.drlx.builder;

import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.Iterator;
import java.util.Map;

import org.drools.base.base.ValueResolver;
import org.drools.base.reteoo.BaseTuple;
import org.drools.base.rule.Declaration;
import org.drools.base.rule.accessor.DataProvider;
import org.mvel3.Evaluator;
import org.kie.api.runtime.rule.FactHandle;

public class FromExpressionDataProvider implements DataProvider, EvaluatorSink {

    private final String expression;
    private Declaration[] declarations;
    private Evaluator<Map<String, Object>, Void, Object> evaluator;

    public FromExpressionDataProvider(String expression,
                                      Evaluator<Map<String, Object>, Void, Object> evaluator,
                                      Declaration... declarations) {
        this.expression = expression;
        this.evaluator = evaluator;
        this.declarations = declarations;
    }

    @Override
    public Declaration[] getRequiredDeclarations() {
        return declarations;
    }

    @Override
    public Object createContext() {
        return null;
    }

    @Override
    public Iterator getResults(BaseTuple tuple, ValueResolver valueResolver,
                               Object providerContext) {
        Object result = evaluate(tuple, valueResolver);
        if (result == null) {
            return Collections.emptyIterator();
        }
        if (result instanceof Object[] arr) {
            return Arrays.asList(arr).iterator();
        }
        if (result instanceof Iterator iter) {
            return iter;
        }
        if (result instanceof Iterable<?> iter) {
            return iter.iterator();
        }
        return Collections.singletonList(result).iterator();
    }

    private Object evaluate(BaseTuple tuple, ValueResolver valueResolver) {
        if (evaluator == null) {
            throw new IllegalStateException(
                    "FromExpressionDataProvider evaluator not yet bound: " + expression);
        }
        Map<String, Object> input = new HashMap<>(declarations.length * 2);
        for (Declaration d : declarations) {
            FactHandle fh = tuple.get(d);
            input.put(d.getIdentifier(), fh != null ? d.getValue(null, fh.getObject()) : null);
        }
        return evaluator.eval(input);
    }

    @Override
    public DataProvider clone() {
        Declaration[] cloned = new Declaration[declarations.length];
        for (int i = 0; i < declarations.length; i++) {
            cloned[i] = declarations[i].clone();
        }
        return new FromExpressionDataProvider(expression, evaluator, cloned);
    }

    @Override
    public void replaceDeclaration(Declaration declaration, Declaration resolved) {
        for (int i = 0; i < declarations.length; i++) {
            if (declarations[i].getIdentifier().equals(declaration.getIdentifier())) {
                declarations[i] = resolved;
            }
        }
    }

    @Override
    public boolean isReactive() {
        return false;
    }

    @SuppressWarnings({"unchecked", "rawtypes"})
    @Override
    public void bindEvaluator(Evaluator<?, ?, ?> evaluator) {
        this.evaluator = (Evaluator<Map<String, Object>, Void, Object>) evaluator;
    }
}
```

public FromExpressionDataProvider createFromExpressionLambda(String expression,
                                                              List<BoundVariable> referencedBindings) {
    int counter = lambdaCounter++;

    org.mvel3.transpiler.context.Declaration<?>[] mvelDeclarations =
            referencedBindings.stream()
                    .map(bv -> org.mvel3.transpiler.context.Declaration.of(bv.name(), bv.type()))
                    .toArray(org.mvel3.transpiler.context.Declaration[]::new);

    Declaration[] requiredDeclarations = referencedBindings.stream()
            .map(BoundVariable::declaration)
            .toArray(Declaration[]::new);

    @SuppressWarnings("unchecked")
    Evaluator<Map<String, Object>, Void, Object> preCompiled =
            (Evaluator<Map<String, Object>, Void, Object>) tryLoadPreCompiled(counter, expression, "from expression");
    if (preCompiled != null) {
        return new FromExpressionDataProvider(expression, preCompiled, requiredDeclarations);
    }

    FromExpressionDataProvider deferred = createBatchFromExpression(expression, mvelDeclarations, requiredDeclarations);
    onLambdaCreated(counter, expression);
    return deferred;
}

@SuppressWarnings({"unchecked", "rawtypes"})
private FromExpressionDataProvider createBatchFromExpression(String expression,
                                                              org.mvel3.transpiler.context.Declaration<?>[] mvelDeclarations,
                                                              Declaration[] requiredDeclarations) {
    CompilerParameters<Map<String, Object>, Void, Object> evalInfo =
            (CompilerParameters) MVEL.<Object>map(mvelDeclarations)
                    .<Object>out(Object.class)
                    .expression(expression)
                    .imports(new HashSet<>(imports))
                    .classManager(batchCompiler.getClassManager())
                    .generatedClassName("GeneratorEvaluator__")
                    .build();
    MVELBatchCompiler.LambdaHandle handle = batchCompiler.add(evalInfo);
    FromExpressionDataProvider provider = new FromExpressionDataProvider(expression, null, requiredDeclarations);
    pendingLambdas.add(new PendingLambda(handle, provider));
    return provider;
}
